const supabase = window.supabase.createClient( window.SUPABASE_CONFIG.url, window.SUPABASE_CONFIG.publishableKey ); let products = []; let cart = []; function loadCart() { const saved = localStorage.getItem("cart"); if (!saved) return; try { cart = JSON.parse(saved); } catch (e) { cart = []; } } function saveCart() { localStorage.setItem("cart", JSON.stringify(cart)); } function money(n) { return `S/ ${Number(n || 0).toFixed(2)}`; } async function loadProducts() { try { const { data, error } = await supabase .from("productos") .select("*") .order("nombre"); if (error) { console.error("Error cargando productos:", error); document.querySelector("#products").innerHTML = `
Error al cargar productos: ${error.message}
`; return; } if (!data || data.length === 0) { document.querySelector("#products").innerHTML = `
No hay productos disponibles.
`; return; } products = data.map((p) => ({ id: p.id, name: p.nombre, category: p.categoria, unit: p.unidad, price: Number(p.precio), emoji: p.emoji || "🥬" })); render(); } catch (err) { console.error("Error general:", err); } } function render() { const q = document.querySelector("#search").value.toLowerCase().trim(); const c = document.querySelector("#category").value; const list = products.filter( (p) => (!q || p.name.toLowerCase().includes(q)) && (!c || p.category === c) ); const el = document.querySelector("#products"); el.innerHTML = list.length ? list.map((p) => `
${p.emoji}
${p.name}
Venta por ${p.unit}
${money(p.price)}
`).join("") : `
No encontramos productos.
`; updateCart(); } function add(id) { const p = products.find((x) => x.id === id); if (!p) return; cart.push({ ...p }); saveCart(); updateCart(); } function removeFromCart(index) { cart.splice(index, 1); saveCart(); updateCart(); } function clearCart() { cart = []; saveCart(); updateCart(); } function updateCart() { const cartWrap = document.querySelector("#cart"); const cartPanel = document.querySelector("#cart-panel"); const count = document.querySelector("#count"); const total = document.querySelector("#total"); const items = document.querySelector("#cart-items"); const totalDetail = document.querySelector("#cart-total"); const hasItems = cart.length > 0; cartWrap.hidden = !hasItems; cartPanel.hidden = !hasItems; if (!hasItems) { count.textContent = "0"; total.textContent = money(0); items.innerHTML = ""; totalDetail.textContent = money(0); return; } const subtotal = cart.reduce((sum, p) => sum + Number(p.price || 0), 0); count.textContent = cart.length; total.textContent = money(subtotal); totalDetail.textContent = money(subtotal); items.innerHTML = cart.map((p, i) => `
${p.emoji} ${p.name} ${money(p.price)}
`).join(""); } async function showOrder() { if (!cart.length) { alert("El carrito está vacío."); return; } const total = cart.reduce((sum, p) => sum + Number(p.price || 0), 0); const text = cart .map((p, i) => `${i + 1}. ${p.emoji} ${p.name} — ${money(p.price)}`) .join("\n"); const mensaje = `📋 PEDIDO VERDULERÍA TERÁN\n\n${text}\n\n💰 Total: ${money(total)}\n\n¿Deseas confirmar el pedido?`; if (confirm(mensaje)) { await guardarPedido(total, text); } } async function guardarPedido(total, detalle) { try { const { error } = await supabase.from("pedidos").insert([ { items: cart.length, total: Number(total), detalle, estado: "pendiente", fecha: new Date().toISOString() } ]); if (error) { console.error("Error guardando pedido:", error); alert("No se pudo guardar el pedido. Revisa la tabla 'pedidos' en Supabase."); return; } alert("✅ Pedido guardado correctamente. Nos contactaremos pronto."); clearCart(); } catch (err) { console.error("Error general al guardar pedido:", err); alert("Hubo un error al procesar el pedido."); } } document.querySelector("#search").addEventListener("input", render); document.querySelector("#category").addEventListener("change", render); document.querySelector("#clear-cart").addEventListener("click", clearCart); loadCart(); loadProducts(); if ("serviceWorker" in navigator) { navigator.serviceWorker.register("./sw.js").catch(() => {}); } updateCart();
