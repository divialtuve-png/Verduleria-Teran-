const supabase = window.supabase.createClient( window.SUPABASE_CONFIG.url, window.SUPABASE_CONFIG.publishableKey ); let products = []; let cart = []; let isAdmin = false; function loadCart() { const saved = localStorage.getItem("cart"); if (saved) { try { cart = JSON.parse(saved); } catch (e) { cart = []; } } } function saveCart() { localStorage.setItem("cart", JSON.stringify(cart)); } async function loadProducts() { try { const { data, error } = await supabase .from("productos") .select("*") .order("nombre"); if (error) { console.error("Error cargando productos:", error); document.querySelector("#products").innerHTML = `
Error al cargar productos: ${error.message}
`; return; } if (!data || data.length === 0) { document.querySelector("#products").innerHTML = `
No hay productos disponibles
`; return; } products = data.map(p => ({ id: p.id, name: p.nombre, category: p.categoria, unit: p.unidad, price: Number(p.precio), emoji: p.emoji || "🥬", activo: p.activo !== false, stock: p.stock || null })); render(); } catch (err) { console.error("Error:", err); } } function money(n) { return `S/ ${Number(n).toFixed(2)}`; } function render() { const q = document.querySelector("#search").value.toLowerCase().trim(); const c = document.querySelector("#category").value; const list = products.filter( p => (!q || p.name.toLowerCase().includes(q)) && (!c || p.category === c) ); const el = document.querySelector("#products"); el.innerHTML = list.length ? list.map(p => `
${p.emoji}
${p.name}
Venta por ${p.unit}
${money(p.price)}
`).join("") : `
No encontramos productos.
`; updateCart(); } function add(id) { const p = products.find(x => x.id === id); if (p) { cart.push({ ...p }); saveCart(); updateCart(); } } function removeFromCart(index) { cart.splice(index, 1); saveCart(); updateCart(); } function updateCart() { const cartEl = document.querySelector("#cart"); const countEl = document.querySelector("#count"); const totalEl = document.querySelector("#total"); const itemsEl = document.querySelector("#cart-items"); const totalDetalleEl = document.querySelector("#cart-total"); cartEl.hidden = !cart.length; if (!cart.length) { itemsEl.innerHTML = ""; totalDetalleEl.textContent = money(0); countEl.textContent = "0"; totalEl.textContent = money(0); return; } countEl.textContent = cart.length; const total = cart.reduce((s, p) => s + p.price, 0); totalEl.textContent = money(total); itemsEl.innerHTML = cart.map((p, i) => `
${p.emoji} ${p.name} ${money(p.price)}
`).join(""); totalDetalleEl.textContent = money(total); } async function showOrder() { if (!cart.length) { alert("El carrito está vacío"); return; } const total = cart.reduce((s, p) => s + p.price, 0); const text = cart .map((p, i) => `${i + 1}. ${p.emoji} ${p.name} — ${money(p.price)}`) .join("\n"); const mensaje = `📋 PEDIDO VERDULERÍA TERÁN\n\n${text}\n\n💰 Total: ${money(total)}\n\n¿Deseas confirmar el pedido?`; if (confirm(mensaje)) { await guardarPedido(total, text); } } async function guardarPedido(total, detalle) { try { const { error } = await supabase .from("pedidos") .insert([ { items: cart.length, total: total, detalle: detalle, estado: "pendiente", fecha: new Date().toISOString() } ]); if (error) { console.error("Error guardando pedido:", error); alert("Error al guardar el pedido: " + error.message); return; } alert("✅ Pedido guardado correctamente!\n\nNos pondremos en contacto pronto."); cart = []; saveCart(); updateCart(); } catch (err) { console.error("Error:", err); } } document.querySelector("#search").addEventListener("input", render); document.querySelector("#category").addEventListener("change", render); loadCart(); loadProducts(); if ("serviceWorker" in navigator) { navigator.serviceWorker .register("./sw.js") .catch(() => {}); }
